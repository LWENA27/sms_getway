// STEP 1: Add import at top of MainActivity.kt
// -----------------------------------------------------------------------------
import com.lwenatech.sms_gateway.services.MarketingMethodChannelHandler

// STEP 2: Add marketing channel constant
// -----------------------------------------------------------------------------
// Add this line after: private val CHANNEL = "com.lwenatech.sms_gateway/sms"
private val MARKETING_CHANNEL = "com.lwenatech.sms_gateway/marketing"

// STEP 3: Register marketing channel in configureFlutterEngine()
// -----------------------------------------------------------------------------
// Add these lines at the END of configureFlutterEngine() method, 
// AFTER the existing MethodChannel.setMethodCallHandler block:

val marketingChannel = MethodChannel(
    flutterEngine.dartExecutor.binaryMessenger, 
    MARKETING_CHANNEL
)
MarketingMethodChannelHandler.register(marketingChannel, applicationContext)
